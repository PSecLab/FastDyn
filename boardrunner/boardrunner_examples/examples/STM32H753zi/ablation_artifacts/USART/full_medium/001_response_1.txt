### 1. High-Level Summary
USART3 is behaving like an interrupt-driven serial port used for bidirectional byte-stream communication, most likely with simple half-duplex phases: firmware enables TXE interrupts to push outgoing bytes, then switches to TC interrupt to detect end-of-transmission, and later enables RXNE plus error interrupt handling to receive a response byte. The trace shows stable idle status bits, immediate transmit readiness, occasional received data in RDR, and IRQ 39 being used for TX-ready / TX-complete / RX-available events. A PTY-backed host interface is the best fit so transmitted bytes are observable and host-supplied bytes can drive RXNE interrupts.

### 2. Register Analysis
- **CR1 (0x00)**  
  Main USART enable/control register. Observed values strongly match:
  - `0x0D` = `UE|RE|TE`
  - `0x8D` = `UE|RE|TE|TXEIE`
  - `0x4D` = `UE|RE|TE|TCIE`
  - `0x2D` = `UE|RE|TE|RXNEIE`
- **CR2 (0x04)**  
  Control register 2. Always observed as `0x0`, so modeled as plain stateful storage.
- **CR3 (0x08)**  
  Control register 3. Observed `0x0` and `0x1`; bit 0 is consistent with `EIE` (error interrupt enable). Modeled as stateful storage with interrupt masking influence.
- **ISR (0x1C)**  
  Status register. Dominant values:
  - `0x6000D0` = `TEACK|REACK|TXE|TC|IDLE`
  - `0x6000F0` = above + `RXNE`
  - occasional `0x6010D0 / 0x6010F0` = same plus transient bit `0x1000` seen around receive activity
- **RDR (0x24)**  
  Receive data register. Read when ISR shows RXNE set.
- **TDR (0x28)**  
  Transmit data register. Firmware writes outgoing bytes here during IRQ handling.
- **PRESC (0x2C)**  
  Mentioned as stateful by analysis though not shown in visible trace excerpt; modeled as ordinary storage.
- **ICR (0x20)**  
  Not explicitly shown in trace, but useful for standards-compliant clearing of sticky error/status bits.

### 3. C Device Model Source Code
```c
// Device Model for USART3
//
// Inferred Register Functions:
//   0x00 CR1   - USART enable + interrupt enables (UE/RE/TE/TXEIE/TCIE/RXNEIE)
//   0x04 CR2   - Stateful storage, observed as 0
//   0x08 CR3   - Error interrupt enable in bit 0 (EIE), otherwise stateful
//   0x1C ISR   - Status: IDLE/TXE/TC/TEACK/REACK always high when enabled; RXNE when host byte queued
//   0x20 ICR   - Clears sticky error/status bits
//   0x24 RDR   - Receive data register
//   0x28 TDR   - Transmit data register (writes forwarded to PTY)
//   0x2C PRESC - Stateful storage
//
// Host interaction:
//   - TDR writes are sent to a host PTY.
//   - Host PTY input is polled and becomes USART RX data in RDR.
//
// IRQ:
//   - USART3 NVIC IRQn = 39, so qemu_plugin_raise_irq uses 55.

#include <device.h>
#include <boardrunner/vio.h>

#include <stdint.h>
#include <stdbool.h>
#include <string.h>

#define USART3_BASE         0x40004800ULL
#define USART3_IRQ_VECTOR   55

/* Register offsets */
#define USART_CR1_OFF       0x00
#define USART_CR2_OFF       0x04
#define USART_CR3_OFF       0x08
#define USART_ISR_OFF       0x1C
#define USART_ICR_OFF       0x20
#define USART_RDR_OFF       0x24
#define USART_TDR_OFF       0x28
#define USART_PRESC_OFF     0x2C

/* CR1 bits used by the trace */
#define USART_CR1_UE                    (1u << 0)
#define USART_CR1_RE                    (1u << 2)
#define USART_CR1_TE                    (1u << 3)
#define USART_CR1_IDLEIE                (1u << 4)
#define USART_CR1_RXNEIE_RXFNEIE        (1u << 5)
#define USART_CR1_TCIE                  (1u << 6)
#define USART_CR1_TXEIE_TXFNFIE         (1u << 7)

/* CR3 bits used by the trace */
#define USART_CR3_EIE                   (1u << 0)

/* ISR bits */
#define USART_ISR_PE                    (1u << 0)
#define USART_ISR_FE                    (1u << 1)
#define USART_ISR_NE                    (1u << 2)
#define USART_ISR_ORE                   (1u << 3)
#define USART_ISR_IDLE                  (1u << 4)
#define USART_ISR_RXNE_RXFNE            (1u << 5)
#define USART_ISR_TC                    (1u << 6)
#define USART_ISR_TXE_TXFNF             (1u << 7)
#define USART_ISR_EOBF                  (1u << 12)   /* used as transient auxiliary status seen in trace */
#define USART_ISR_TEACK                 (1u << 21)
#define USART_ISR_REACK                 (1u << 22)

/* ICR bits corresponding to sticky flags we model */
#define USART_ICR_PECF                  (1u << 0)
#define USART_ICR_FECF                  (1u << 1)
#define USART_ICR_NECF                  (1u << 2)
#define USART_ICR_ORECF                 (1u << 3)
#define USART_ICR_IDLECF                (1u << 4)
#define USART_ICR_TCCF                  (1u << 6)
#define USART_ICR_EOBCF                 (1u << 12)

#define USART3_RX_FIFO_SIZE             256
#define USART3_POLL_PERIOD_NS           1000000ULL   /* 1 ms */

typedef struct {
    uint32_t cr1;
    uint32_t cr2;
    uint32_t cr3;
    uint32_t presc;

    uint32_t sticky_isr;       /* PE/FE/NE/ORE etc. */
    uint8_t  transient_eob_reads; /* short-lived 0x1000 marker seen around some RX activity */

    uint8_t  rx_fifo[USART3_RX_FIFO_SIZE];
    unsigned rx_head;
    unsigned rx_tail;
    unsigned rx_count;

    int      pty_fd;
    uint64_t poll_timer;
} USART3State;

static USART3State g_usart3;

static bool usart3_rx_fifo_push(USART3State *s, uint8_t v)
{
    if (s->rx_count >= USART3_RX_FIFO_SIZE) {
        return false;
    }

    s->rx_fifo[s->rx_tail] = v;
    s->rx_tail = (s->rx_tail + 1) % USART3_RX_FIFO_SIZE;
    s->rx_count++;
    return true;
}

static bool usart3_rx_fifo_pop(USART3State *s, uint8_t *out)
{
    if (s->rx_count == 0) {
        return false;
    }

    *out = s->rx_fifo[s->rx_head];
    s->rx_head = (s->rx_head + 1) % USART3_RX_FIFO_SIZE;
    s->rx_count--;
    return true;
}

static uint32_t usart3_compute_isr(USART3State *s)
{
    uint32_t isr = s->sticky_isr;

    if ((s->cr1 & USART_CR1_UE) && (s->cr1 & USART_CR1_TE)) {
        isr |= USART_ISR_TXE_TXFNF;
        isr |= USART_ISR_TC;
        isr |= USART_ISR_TEACK;
    }

    if ((s->cr1 & USART_CR1_UE) && (s->cr1 & USART_CR1_RE)) {
        isr |= USART_ISR_IDLE;
        isr |= USART_ISR_REACK;

        if (s->rx_count > 0) {
            isr |= USART_ISR_RXNE_RXFNE;
        }
    }

    if (s->transient_eob_reads > 0) {
        isr |= USART_ISR_EOBF;
    }

    return isr;
}

static void usart3_raise_irq_if_needed(USART3State *s)
{
    uint32_t isr = usart3_compute_isr(s);
    bool fire = false;

    if ((isr & USART_ISR_RXNE_RXFNE) && (s->cr1 & USART_CR1_RXNEIE_RXFNEIE)) {
        fire = true;
    }

    if ((isr & USART_ISR_TXE_TXFNF) && (s->cr1 & USART_CR1_TXEIE_TXFNFIE)) {
        fire = true;
    }

    if ((isr & USART_ISR_TC) && (s->cr1 & USART_CR1_TCIE)) {
        fire = true;
    }

    if ((isr & USART_ISR_IDLE) && (s->cr1 & USART_CR1_IDLEIE)) {
        fire = true;
    }

    if ((isr & (USART_ISR_PE | USART_ISR_FE | USART_ISR_NE | USART_ISR_ORE)) &&
        (s->cr3 & USART_CR3_EIE)) {
        fire = true;
    }

    if (fire) {
        qemu_plugin_raise_irq(USART3_IRQ_VECTOR, false);
    }
}

static void usart3_poll_host_input(USART3State *s)
{
    uint8_t ch;
    int ret;
    bool new_data = false;

    if (s->pty_fd < 0) {
        return;
    }

    for (;;) {
        ret = api_pty_read_nonblock(s->pty_fd, &ch);
        if (ret <= 0) {
            break;
        }

        if (!usart3_rx_fifo_push(s, ch)) {
            s->sticky_isr |= USART_ISR_ORE;
        } else {
            new_data = true;
            /*
             * Trace occasionally shows ISR bit 0x1000 around receive activity.
             * Keep a short-lived marker so ISR can expose 0x6010F0/0x6010D0
             * without inventing a large protocol-specific state machine.
             */
            s->transient_eob_reads = 2;
        }
    }

    if (new_data || (s->sticky_isr & USART_ISR_ORE)) {
        usart3_raise_irq_if_needed(s);
    }
}

static void usart3_poll_timer_cb(void *opaque)
{
    USART3State *s = (USART3State *)opaque;
    usart3_poll_host_input(s);
}

// This function will emulate all device reads
uint64_t usart3_read(void *opaque, hwaddr addr, unsigned size)
{
    USART3State *s = (USART3State *)opaque;
    hwaddr offset = addr - USART3_BASE;
    uint32_t val = 0;

    (void)size;

    /* Poll synchronously as well as from timer so tight firmware polling can see host input. */
    usart3_poll_host_input(s);

    switch (offset) {
    case USART_CR1_OFF:
        val = s->cr1;
        break;

    case USART_CR2_OFF:
        val = s->cr2;
        break;

    case USART_CR3_OFF:
        val = s->cr3;
        break;

    case USART_ISR_OFF:
        val = usart3_compute_isr(s);
        if (s->transient_eob_reads > 0) {
            s->transient_eob_reads--;
        }
        break;

    case USART_RDR_OFF: {
        uint8_t ch = 0;
        if (usart3_rx_fifo_pop(s, &ch)) {
            val = ch;
            if (s->rx_count > 0) {
                usart3_raise_irq_if_needed(s);
            }
        } else {
            val = 0;
        }
        break;
    }

    case USART_PRESC_OFF:
        val = s->presc;
        break;

    case USART_TDR_OFF:
        val = 0;
        break;

    case USART_ICR_OFF:
        val = 0;
        break;

    default:
        val = 0;
        break;
    }

    return val;
}

// This function will emulate all device writes
void usart3_write(void *opaque, hwaddr addr, uint64_t value, unsigned size)
{
    USART3State *s = (USART3State *)opaque;
    hwaddr offset = addr - USART3_BASE;
    uint32_t v = (uint32_t)value;

    (void)size;

    switch (offset) {
    case USART_CR1_OFF:
        s->cr1 = v;
        /*
         * Enabling TXEIE/TCIE with TX path already empty/complete should
         * immediately make the IRQ observable, matching the trace pattern.
         */
        usart3_raise_irq_if_needed(s);
        break;

    case USART_CR2_OFF:
        s->cr2 = v;
        break;

    case USART_CR3_OFF:
        s->cr3 = v;
        usart3_raise_irq_if_needed(s);
        break;

    case USART_TDR_OFF:
        /*
         * Forward the transmitted byte to the host-visible PTY.
         * Only the low byte is meaningful for the observed use.
         */
        if (s->pty_fd >= 0) {
            api_pty_write_req(s->pty_fd, (uint8_t)(v & 0xFF));
        }

        /*
         * The observed firmware expects transmit-ready / transmit-complete
         * interrupts to continue promptly after writes.
         */
        usart3_raise_irq_if_needed(s);
        break;

    case USART_ICR_OFF:
        if (v & USART_ICR_PECF) {
            s->sticky_isr &= ~USART_ISR_PE;
        }
        if (v & USART_ICR_FECF) {
            s->sticky_isr &= ~USART_ISR_FE;
        }
        if (v & USART_ICR_NECF) {
            s->sticky_isr &= ~USART_ISR_NE;
        }
        if (v & USART_ICR_ORECF) {
            s->sticky_isr &= ~USART_ISR_ORE;
        }
        if (v & USART_ICR_EOBCF) {
            s->transient_eob_reads = 0;
        }
        /* IDLE/TC are computed hardware state in this minimal model. */
        break;

    case USART_PRESC_OFF:
        s->presc = v;
        break;

    case USART_RDR_OFF:
    case USART_ISR_OFF:
    default:
        /* Read-only or unimplemented register in this minimal model. */
        break;
    }
}

// MUST return pointer to state — framework passes it as opaque to _read/_write
void* usart3_init(ConfigSection* model_info)
{
    (void)model_info;

    memset(&g_usart3, 0, sizeof(g_usart3));

    g_usart3.pty_fd = api_pty_fd_gen();
    g_usart3.poll_timer = qemu_plugin_timer_new_period_ns(usart3_poll_timer_cb,
                                                          &g_usart3,
                                                          USART3_POLL_PERIOD_NS);

    dev_debug("USART3: initialized with PTY-backed serial emulation");
    return &g_usart3;
}
```
