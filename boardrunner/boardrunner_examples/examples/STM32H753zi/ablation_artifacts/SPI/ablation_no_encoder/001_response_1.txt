### 1. High-Level Summary
SPI1 on this STM32H753x trace is being used as a simple 8-bit SPI master. Firmware configures CFG1/CFG2, sets a transfer size of 2 bytes in CR2, enables the peripheral via CR1, starts a master transaction with CSTART, writes two bytes through TXDR, polls SR for TX/RX/EOT conditions, reads received bytes from RXDR, clears completion flags through IFCR, and then disables the peripheral. The model below implements that stateful behavior, including a minimal but trace-consistent SR model and full-duplex transfers via the framework SPI bus API.

### 2. Register Analysis
- **0x40013000 — CR1**
  - Written as `0x00000000`, later `0x00000001` (SPE), then `0x00000201` (SPE + CSTART).
  - Read back as `0x00000001` after the transfer, indicating `CSTART` is hardware-cleared on completion.
- **0x40013004 — CR2**
  - Written with `0x00000002`, strongly indicating `TSIZE = 2` bytes.
- **0x40013008 — CFG1**
  - Written with `0x70070007`; appears to configure data size / FIFO thresholds / prescaler fields.
  - Read back unchanged.
- **0x4001300C — CFG2**
  - Written first with `0x20400000`, later `0xA0400000`; read back unchanged.
  - Represents master mode / polarity / phase / NSS-related configuration.
- **0x40013010 — IER**
  - Read and written as `0x00000000`; interrupts are not actively used in this trace.
- **0x40013014 — SR**
  - Key polled status register.
  - Important observed values:
    - `0x00020002` before data transfer starts.
    - `0x00022007` after first TX byte, with RX data available.
    - `0x0000301F` after final TX byte, showing transfer completion and RX data pending.
    - `0x0000101A` after final RX read, showing completion flags still set.
- **0x40013018 — IFCR**
  - Read as `0x00000000`.
  - Written with `0x00000008` then `0x00000010`, consistent with clearing EOT/TXTF-style completion flags.
- **0x40013020 — TXDR**
  - Byte writes: `0xD0`, then `0x00`.
- **0x40013030 — RXDR**
  - Byte reads return `0xFF`, then `0x60`.
- **0x40013050 — I2SCFGR**
  - Read/write as `0x00000000`; I2S mode unused here.

### 3. C Device Model Source Code
```c
// Device Model for SPI1
//
// Inferred Register Functions:
//   0x00 CR1     - control register 1 (SPE, CSTART observed)
//   0x04 CR2     - control register 2 (TSIZE observed)
//   0x08 CFG1    - configuration register 1
//   0x0C CFG2    - configuration register 2
//   0x10 IER     - interrupt enable register
//   0x14 SR      - status register
//   0x18 IFCR    - interrupt/flag clear register
//   0x20 TXDR    - transmit data register
//   0x30 RXDR    - receive data register
//   0x40 CRCPOLY - CRC polynomial register
//   0x44 TXCRC   - TX CRC register
//   0x48 RXCRC   - RX CRC register
//   0x4C UDRDR   - underrun data register
//   0x50 I2SCFGR - I2S configuration register

#include <device.h>
#include <boardrunner/vio.h>
#include <stdint.h>
#include <stdbool.h>
#include <string.h>

#define SPI1_BASE 0x40013000ULL

#define SPI1_CR1_OFFSET      0x00
#define SPI1_CR2_OFFSET      0x04
#define SPI1_CFG1_OFFSET     0x08
#define SPI1_CFG2_OFFSET     0x0C
#define SPI1_IER_OFFSET      0x10
#define SPI1_SR_OFFSET       0x14
#define SPI1_IFCR_OFFSET     0x18
#define SPI1_TXDR_OFFSET     0x20
#define SPI1_RXDR_OFFSET     0x30
#define SPI1_CRCPOLY_OFFSET  0x40
#define SPI1_TXCRC_OFFSET    0x44
#define SPI1_RXCRC_OFFSET    0x48
#define SPI1_UDRDR_OFFSET    0x4C
#define SPI1_I2SCFGR_OFFSET  0x50

/* Observed CR1 bits */
#define SPI1_CR1_SPE         (1u << 0)
#define SPI1_CR1_CSTART      (1u << 9)

/*
 * Minimal SR bit model chosen to match the trace:
 *   bit 0  RXP   - RX data available
 *   bit 1  TXP   - TX data can be written
 *   bit 2  DXP   - duplex exchange possible (RXP|TXP)
 *   bit 3  EOT   - end of transfer
 *   bit 4  TXTF  - transfer filled/completed
 *   bit 12 TXC   - transmission complete
 *   bit 13 RXPLVL/non-empty indication
 *   bits 31:16   - trace-consistent active TSIZE mirror while transfer active
 */
#define SPI1_SR_RXP          (1u << 0)
#define SPI1_SR_TXP          (1u << 1)
#define SPI1_SR_DXP          (1u << 2)
#define SPI1_SR_EOT          (1u << 3)
#define SPI1_SR_TXTF         (1u << 4)
#define SPI1_SR_TXC          (1u << 12)
#define SPI1_SR_RXPLVL       (1u << 13)

#define SPI1_RX_FIFO_SIZE    16

typedef struct {
    SPIBus bus;

    uint32_t cr1_base;   /* software-controlled CR1 bits except live SPE/CSTART */
    uint32_t cr2;
    uint32_t cfg1;
    uint32_t cfg2;
    uint32_t ier;
    uint32_t crcpoly;
    uint32_t txcrc;
    uint32_t rxcrc;
    uint32_t udrdr;
    uint32_t i2scfgr;

    bool spe;
    bool cstart;
    bool eot;
    bool txtf;
    bool txc;

    uint16_t tsize;
    uint16_t transferred;

    uint8_t rx_fifo[SPI1_RX_FIFO_SIZE];
    int rx_fifo_len;

    int default_cs;
    bool cs_asserted;
} SPI1State;

static SPI1State g_spi1;

static void spi1_debug(const char *msg) {
    dev_debug((char *)msg);
}

static void spi1_set_cs(SPI1State *s, bool active) {
    if (s->bus.Slaves.num_slaves <= 0) {
        s->cs_asserted = false;
        return;
    }

    if (s->cs_asserted == active) {
        return;
    }

    api_spi_set_cs(&s->bus, s->default_cs, active ? 0 : 1);
    s->cs_asserted = active;
}

static void spi1_rx_fifo_clear(SPI1State *s) {
    s->rx_fifo_len = 0;
}

static void spi1_rx_fifo_push(SPI1State *s, uint8_t v) {
    if (s->rx_fifo_len >= SPI1_RX_FIFO_SIZE) {
        /* Minimal model: drop oldest when full */
        memmove(&s->rx_fifo[0], &s->rx_fifo[1], SPI1_RX_FIFO_SIZE - 1);
        s->rx_fifo_len = SPI1_RX_FIFO_SIZE - 1;
    }
    s->rx_fifo[s->rx_fifo_len++] = v;
}

static uint8_t spi1_rx_fifo_pop(SPI1State *s) {
    uint8_t v = 0;
    if (s->rx_fifo_len <= 0) {
        return 0;
    }

    v = s->rx_fifo[0];
    if (s->rx_fifo_len > 1) {
        memmove(&s->rx_fifo[0], &s->rx_fifo[1], s->rx_fifo_len - 1);
    }
    s->rx_fifo_len--;
    return v;
}

static void spi1_begin_transfer(SPI1State *s) {
    s->cstart = true;
    s->eot = false;
    s->txtf = false;
    s->txc = false;
    s->transferred = 0;
    spi1_rx_fifo_clear(s);
    spi1_set_cs(s, true);
}

static void spi1_finish_transfer(SPI1State *s) {
    s->cstart = false;   /* trace shows CR1 reads back with CSTART cleared */
    s->eot = true;
    s->txtf = true;
    s->txc = true;
    spi1_set_cs(s, false);
}

static uint32_t spi1_compose_cr1(SPI1State *s) {
    uint32_t v = s->cr1_base;
    if (s->spe) {
        v |= SPI1_CR1_SPE;
    } else {
        v &= ~SPI1_CR1_SPE;
    }

    if (s->cstart) {
        v |= SPI1_CR1_CSTART;
    } else {
        v &= ~SPI1_CR1_CSTART;
    }

    return v;
}

static uint32_t spi1_compose_sr(SPI1State *s) {
    uint32_t sr = 0;

    if (s->spe) {
        sr |= SPI1_SR_TXP;

        if (s->rx_fifo_len > 0) {
            sr |= SPI1_SR_RXP;
            sr |= SPI1_SR_DXP;
            sr |= SPI1_SR_RXPLVL;
        }

        if (s->eot) {
            sr |= SPI1_SR_EOT;
        }
        if (s->txtf) {
            sr |= SPI1_SR_TXTF;
        }
        if (s->txc) {
            sr |= SPI1_SR_TXC;
        }

        /*
         * Trace-consistent behavior:
         * while transfer is active and not yet complete, the upper halfword
         * shows the programmed TSIZE value (seen as 0x0002xxxx for TSIZE=2).
         */
        if (s->cstart && !s->eot && s->tsize != 0) {
            sr |= ((uint32_t)s->tsize) << 16;
        }
    }

    return sr;
}

static void spi1_transfer_one_byte(SPI1State *s, uint8_t txb) {
    uint32_t rxb = 0xFF;

    if (!s->spe) {
        return;
    }

    if (!s->cstart && s->tsize != 0) {
        return;
    }

    if (s->bus.Slaves.num_slaves > 0) {
        spi1_set_cs(s, true);
        rxb = api_spi_transfer(&s->bus, txb);
    }

    spi1_rx_fifo_push(s, (uint8_t)(rxb & 0xFF));
    s->transferred++;

    if (s->tsize != 0 && s->transferred >= s->tsize) {
        spi1_finish_transfer(s);
    }
}

static uint64_t spi1_read_data_sized(uint32_t reg, unsigned size) {
    switch (size) {
    case 1:
        return reg & 0xFFu;
    case 2:
        return reg & 0xFFFFu;
    default:
        return reg;
    }
}

static uint64_t spi1_read_rxdr(SPI1State *s, unsigned size) {
    uint32_t v = 0;
    unsigned i;

    if (size != 1 && size != 2 && size != 4) {
        size = 1;
    }

    for (i = 0; i < size; i++) {
        v |= ((uint32_t)spi1_rx_fifo_pop(s)) << (i * 8);
    }

    return v;
}

static void spi1_write_txdr(SPI1State *s, uint64_t value, unsigned size) {
    unsigned i;

    if (size != 1 && size != 2 && size != 4) {
        size = 1;
    }

    for (i = 0; i < size; i++) {
        uint8_t b = (uint8_t)((value >> (i * 8)) & 0xFFu);
        spi1_transfer_one_byte(s, b);
    }
}

// This function will emulate all device reads
uint64_t spi1_read(void *opaque, hwaddr addr, unsigned size) {
    SPI1State *s = (SPI1State *)opaque;
    hwaddr offset = addr - SPI1_BASE;

    switch (offset) {
    case SPI1_CR1_OFFSET:
        return spi1_read_data_sized(spi1_compose_cr1(s), size);
    case SPI1_CR2_OFFSET:
        return spi1_read_data_sized(s->cr2, size);
    case SPI1_CFG1_OFFSET:
        return spi1_read_data_sized(s->cfg1, size);
    case SPI1_CFG2_OFFSET:
        return spi1_read_data_sized(s->cfg2, size);
    case SPI1_IER_OFFSET:
        return spi1_read_data_sized(s->ier, size);
    case SPI1_SR_OFFSET:
        return spi1_read_data_sized(spi1_compose_sr(s), size);
    case SPI1_IFCR_OFFSET:
        /* write-only in practice for this firmware */
        return 0;
    case SPI1_TXDR_OFFSET:
        return 0;
    case SPI1_RXDR_OFFSET:
        return spi1_read_rxdr(s, size);
    case SPI1_CRCPOLY_OFFSET:
        return spi1_read_data_sized(s->crcpoly, size);
    case SPI1_TXCRC_OFFSET:
        return spi1_read_data_sized(s->txcrc, size);
    case SPI1_RXCRC_OFFSET:
        return spi1_read_data_sized(s->rxcrc, size);
    case SPI1_UDRDR_OFFSET:
        return spi1_read_data_sized(s->udrdr, size);
    case SPI1_I2SCFGR_OFFSET:
        return spi1_read_data_sized(s->i2scfgr, size);
    default:
        spi1_debug("SPI1: unhandled read");
        return 0;
    }
}

// This function will emulate all device writes
void spi1_write(void *opaque, hwaddr addr, uint64_t value, unsigned size) {
    SPI1State *s = (SPI1State *)opaque;
    hwaddr offset = addr - SPI1_BASE;

    switch (offset) {
    case SPI1_CR1_OFFSET: {
        bool new_spe = (value & SPI1_CR1_SPE) != 0;
        bool start_req = (value & SPI1_CR1_CSTART) != 0;

        /* Keep all other CR1 bits except the two live bits we model explicitly */
        s->cr1_base = ((uint32_t)value) & ~(SPI1_CR1_SPE | SPI1_CR1_CSTART);

        if (!new_spe) {
            s->spe = false;
            s->cstart = false;
            s->eot = false;
            s->txtf = false;
            s->txc = false;
            s->transferred = 0;
            spi1_rx_fifo_clear(s);
            spi1_set_cs(s, false);
            return;
        }

        s->spe = true;

        if (start_req && !s->cstart) {
            spi1_begin_transfer(s);
        }
        break;
    }

    case SPI1_CR2_OFFSET:
        s->cr2 = (uint32_t)value;
        s->tsize = (uint16_t)(s->cr2 & 0xFFFFu);
        break;

    case SPI1_CFG1_OFFSET:
        s->cfg1 = (uint32_t)value;
        break;

    case SPI1_CFG2_OFFSET:
        s->cfg2 = (uint32_t)value;
        break;

    case SPI1_IER_OFFSET:
        s->ier = (uint32_t)value;
        break;

    case SPI1_SR_OFFSET:
        /* status register is hardware-owned in this model */
        break;

    case SPI1_IFCR_OFFSET: {
        uint32_t v = (uint32_t)value;

        if (v & SPI1_SR_EOT) {
            s->eot = false;
        }
        if (v & SPI1_SR_TXTF) {
            s->txtf = false;
        }
        /*
         * TXC is intentionally left sticky here; firmware does not rely on
         * clearing it in this trace.
         */
        break;
    }

    case SPI1_TXDR_OFFSET:
        spi1_write_txdr(s, value, size);
        break;

    case SPI1_RXDR_OFFSET:
        /* read-only */
        break;

    case SPI1_CRCPOLY_OFFSET:
        s->crcpoly = (uint32_t)value;
        break;

    case SPI1_TXCRC_OFFSET:
        s->txcrc = (uint32_t)value;
        break;

    case SPI1_RXCRC_OFFSET:
        s->rxcrc = (uint32_t)value;
        break;

    case SPI1_UDRDR_OFFSET:
        s->udrdr = (uint32_t)value;
        break;

    case SPI1_I2SCFGR_OFFSET:
        s->i2scfgr = (uint32_t)value;
        break;

    default:
        spi1_debug("SPI1: unhandled write");
        break;
    }
}

// MUST return pointer to state — framework passes it as opaque to _read/_write
void* spi1_init(ConfigSection* model_info) {
    int i;

    memset(&g_spi1, 0, sizeof(g_spi1));

    g_spi1.bus = api_spi_init_bus(model_info);
    g_spi1.default_cs = 0;
    if (g_spi1.bus.Slaves.num_slaves > 0) {
        g_spi1.default_cs = g_spi1.bus.Slaves.slave[0].cs_id;
    }

    for (i = 0; i < NUM_CS_LINES; i++) {
        api_spi_set_cs(&g_spi1.bus, i, 1);
    }

    /*
     * Registers are largely reset to zero in the trace before configuration.
     * No stronger reset evidence is needed for this minimal model.
     */
    g_spi1.cr1_base = 0;
    g_spi1.cr2 = 0;
    g_spi1.cfg1 = 0;
    g_spi1.cfg2 = 0;
    g_spi1.ier = 0;
    g_spi1.crcpoly = 0;
    g_spi1.txcrc = 0;
    g_spi1.rxcrc = 0;
    g_spi1.udrdr = 0;
    g_spi1.i2scfgr = 0;

    g_spi1.spe = false;
    g_spi1.cstart = false;
    g_spi1.eot = false;
    g_spi1.txtf = false;
    g_spi1.txc = false;
    g_spi1.tsize = 0;
    g_spi1.transferred = 0;
    g_spi1.cs_asserted = false;
    g_spi1.rx_fifo_len = 0;

    return &g_spi1;
}
```
